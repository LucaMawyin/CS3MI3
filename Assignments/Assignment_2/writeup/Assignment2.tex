\section{}

First, we will represent the general if-else rule:
\begin{align*}
    \frac{t_0\to t'_0}{\text{If }t_0\text{ then }t_1\to\text{If }t'_0\text{ then }t_1}\\
\end{align*}
Then we will represent the rules for True:
\begin{align*}
    \frac{}{\text{If True then }t_1\to t_1}
\end{align*}
and False:
\begin{align*}
    \frac{}{\text{If False then }t_1\to t_1}
\end{align*}

\section{}

\begin{verbatim}

------------------------------
-- RTerm
------------------------------

data RTerm
    = RTrue 
    | RFalse
    | If RTerm RTerm
    deriving (Show, Eq)

------------------------------
-- evalOne
------------------------------

evalOne :: RTerm -> Maybe RTerm

-- Eval true/false does nothing
evalOne RTrue = Nothing
evalOne RFalse = Nothing

-- Eval with if true/false
evalOne (If RTrue t) = Just t
evalOne (If RFalse t) = Just RTrue

-- Eval t0 and decide what to return
evalOne (If t0 t1) = 
    case evalOne t0 of
        Just t0' -> Just (If t0' t1)
        Nothing -> Nothing

------------------------------
-- evalBig
------------------------------

evalBig :: RTerm -> RTerm 
evalBig t = maybe t evalBig (evalOne t)
\end{verbatim}
